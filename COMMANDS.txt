Download dataset from the base dir
set -e
df -h /workspace | tail -1
mkdir -p /workspace/ogbench_data ~/.ogbench/data
for f in cube-triple-play-v0 cube-triple-play-v0-val; do
  curl -fL -A "Mozilla/5.0" -o /workspace/ogbench_data/$f.npz.tmp https://rail.eecs.berkeley.edu/datasets/ogbench/$f.npz
  mv /workspace/ogbench_data/$f.npz.tmp /workspace/ogbench_data/$f.npz
  ln -sf /workspace/ogbench_data/$f.npz ~/.ogbench/data/$f.npz
done
ls -lhL ~/.ogbench/data/
python -c "
import ogbench
env, tr, va = ogbench.make_env_and_datasets('cube-triple-play-singletask-task3-v0')
print('OK', len(tr['terminals']), 'train transitions,', len(va['terminals']), 'val')
"
